cmake_minimum_required(VERSION 3.20)
project(counting_simulator VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(COUNTING_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(COUNTING_MUTATION "none" CACHE STRING "Test-only mutant: none, ignore_multiplicity, skip_integrality, skip_suffix")
set(COUNTING_GMP_ROOT "" CACHE PATH "GMP installation prefix for this target")
if(COUNTING_GMP_ROOT)
  find_path(GMP_INCLUDE_DIR gmp.h PATHS "${COUNTING_GMP_ROOT}/include" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
  find_library(GMP_LIBRARY NAMES gmp PATHS "${COUNTING_GMP_ROOT}/lib" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
else()
  find_path(GMP_INCLUDE_DIR gmp.h HINTS /opt/homebrew/opt/gmp/include)
  find_library(GMP_LIBRARY NAMES gmp HINTS /opt/homebrew/opt/gmp/lib)
endif()
if(NOT GMP_INCLUDE_DIR OR NOT GMP_LIBRARY)
  message(FATAL_ERROR "GMP not found; set COUNTING_GMP_ROOT to the target installation prefix")
endif()
add_library(counting_engine STATIC src/engine.c src/json.c)
target_include_directories(counting_engine PUBLIC include PRIVATE src)
target_include_directories(counting_engine SYSTEM PRIVATE "${GMP_INCLUDE_DIR}")
target_link_libraries(counting_engine PUBLIC "${GMP_LIBRARY}")
target_compile_options(counting_engine PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(COUNTING_MUTATION STREQUAL "ignore_multiplicity")
  target_compile_definitions(counting_engine PRIVATE COUNTING_MUTATE_MULTIPLICITY)
elseif(COUNTING_MUTATION STREQUAL "skip_integrality")
  target_compile_definitions(counting_engine PRIVATE COUNTING_MUTATE_INTEGRALITY)
elseif(COUNTING_MUTATION STREQUAL "skip_suffix")
  target_compile_definitions(counting_engine PRIVATE COUNTING_MUTATE_SUFFIX)
elseif(NOT COUNTING_MUTATION STREQUAL "none")
  message(FATAL_ERROR "Unknown COUNTING_MUTATION")
endif()
if(COUNTING_SANITIZERS AND NOT EMSCRIPTEN)
  target_compile_options(counting_engine PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(counting_engine PUBLIC -fsanitize=address,undefined)
endif()
if(EMSCRIPTEN)
  add_executable(engine src/wasm.c)
  target_link_libraries(engine PRIVATE counting_engine)
  set_target_properties(engine PROPERTIES SUFFIX ".mjs")
  target_link_options(engine PRIVATE --no-entry
    "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sENVIRONMENT=web,worker,node"
    "-sALLOW_MEMORY_GROWTH=1" "-sSTACK_SIZE=1048576"
    "-sEXPORTED_FUNCTIONS=['_sim_create','_sim_step','_sim_snapshot','_sim_summary','_sim_destroy','_sim_free_string','_sim_last_error','_sim_audit_json','_malloc','_free']"
    "-sEXPORTED_RUNTIME_METHODS=['ccall','UTF8ToString','HEAPU8']")
else()
  add_executable(counting src/main.c)
  target_include_directories(counting PRIVATE src)
  target_link_libraries(counting PRIVATE counting_engine)
endif()
